package com.truesoft.supabase.push;

import androidx.annotation.NonNull;

import com.google.firebase.messaging.FirebaseMessagingService;
import com.google.firebase.messaging.RemoteMessage;

/**
 * TrueSoft Supabase SDK — FCM 등록과 받은 알림을 처리한다. 알림(notification) 메시지는 앱이 뒤에 있거나 꺼져 있으면
 * FCM 이 알아서 띄우고 여기로 오지 않는다. 여기로 오는 것은 앱이 앞에 떠 있을 때 온 알림뿐이다 —
 * 기기에는 띄우지 않고 게임에 넘긴다(C# 이 TrueSoftPush.pollReceived 로 꺼내 Supabase.OnPushReceived 로 알린다).
 */
public class TrueSoftPushService extends FirebaseMessagingService {
    @Override
    public void onRegistered(@NonNull String installationId) {
        TrueSoftPush.onRegistered(getApplicationContext(), installationId);
    }

    @Override
    public void onMessageReceived(@NonNull RemoteMessage message) {
        RemoteMessage.Notification n = message.getNotification();
        TrueSoftPush.onReceived(n != null ? n.getTitle() : null, n != null ? n.getBody() : null, message.getData());
    }
}
